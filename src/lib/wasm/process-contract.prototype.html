Proposed public interface

export async function createProcessor(
  options?: { wasm?: InitInput }
): Promise<Processor>;

export interface Processor {
  process(request: ProcessRequest): IndexedImage;
}

export type ProcessRequest = {
  source: Rgba8Image;
  palette: readonly PaletteEntry[];
  recipe: RecipeV1;
};

export type RecipeV1 = {
  version: 1;
  output: { width: number; height: number; resize: Resize };
  alpha: AlphaPolicy;
  match: MatchPolicy;
  dither: DitherPolicy;
};

export type IndexedImage = {
  width: number;
  height: number;
  indices: Uint8Array;
  palette: { rgba: Uint8Array; transparentIndex: number | null };
  warnings: readonly ProcessWarning[];
};

export class DitheretteError extends Error {
  code: ErrorCode;
  path?: string;
  details?: Readonly<Record<string, unknown>>;
}

What stays hidden

Raw wasm-bindgen exports, JSON serialization, palette packing, stage order, scratch memory, caches, tiling, thread policy, benchmarks, and algorithm dispatch. There is no public stage graph.
